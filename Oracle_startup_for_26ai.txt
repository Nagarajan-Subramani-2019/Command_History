Tablespace and schema creation and drop :
-----------------------------------------------
for i in UI_DATA_SCHEMA
do
echo $i
sqlplus -s sys/welcome1@127.0.0.1:11521/FREEPDB1 as sysdba <<EOF
 
SET HEADING OFF
SET FEEDBACK OFF
CREATE TABLESPACE $i DATAFILE '/opt/oracle/oradata/FREE/FREEPDB1/$i.dbf' SIZE 50M AUTOEXTEND ON MAXSIZE 100M;
CREATE USER $i IDENTIFIED BY welcome1 DEFAULT TABLESPACE $i TEMPORARY TABLESPACE TEMP;
ALTER USER $i QUOTA UNLIMITED ON $i;
GRANT CREATE SESSION TO $i;
GRANT CREATE TABLE TO $i;
GRANT CREATE SEQUENCE TO $i;
GRANT CREATE VIEW TO $i;
GRANT CREATE PROCEDURE TO $i;
GRANT CREATE SYNONYM TO $i;
GRANT CREATE TRIGGER TO $i;
EXIT;
EOF
done

DROP USER $i CASCADE;
DROP TABLESPACE $i INCLUDING CONTENTS AND DATAFILES CASCADE CONSTRAINTS;
