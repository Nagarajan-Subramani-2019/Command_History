   # --- 1. WSL & SYSTEM TROUBLESHOOTING ---

# List installed WSL distributions and status
wsl -l -v

# Force shutdown all running WSL distributions
wsl --shutdown

# Restart WSL Windows services (Run in PowerShell as Administrator)
taskkill /f /im wslservice.exe
taskkill /f /im wslhost.exe
net stop LxssManager
net start LxssManager

# Set your default interactive WSL distribution
wsl --setdefault Ubuntu

# Fix broken Docker socket link inside WSL terminal
sudo rm -f /var/run/docker.sock
sudo ln -s /mnt/wsl/rancher-desktop/docker.sock /var/run/docker.sock
unset DOCKER_HOST


# --- 2. DOCKER IMAGE MANAGEMENT ---

# List local Docker images
docker images

# Pull the Oracle Database Free image
docker pull container-registry.oracle.com/database/free:latest

# Log into external registry (if authentication is required)
docker login container-registry.oracle.com


# --- 3. RUNNING & MANAGING THE ORACLE CONTAINER ---

# Run Oracle container with persistent volume on C: drive
docker run -d \
  --name oracle-free \
  -p 1521:1521 -p 5500:5500 \
  -e ORACLE_PWD=YourSecurePassword123! \
  -v /mnt/c/oracle_data:/opt/oracle/oradata \
  container-registry.oracle.com/database/free:latest

# Track initialization logs (Wait for "DATABASE IS READY TO USE!")
docker logs -f oracle-free

# Stop the Oracle container
docker stop oracle-free

# Force kill the Oracle container
docker kill oracle-free

# Start the stopped container
docker start oracle-free

# Start and follow logs immediately
docker start -a oracle-free

# Restart the container
docker restart oracle-free

# List only running containers
docker ps

# List all containers (including stopped ones)
docker ps -a

# Force remove the container (leaves volume data safe on C:\oracle_data)
docker rm -f oracle-free
